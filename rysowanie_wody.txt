void WaterCanvaObject::render_geometry(SDL_Renderer* renderer, Vector2D<int> origin) const {
    std::vector<SDL_Vertex> vertices;
    
    int left = main_r_.get_left();
    int right = main_r_.get_right();
    int bottom = main_r_.get_bottom();
    int surface_y_base = main_r_.get_y() + 20;

    int step = 4; // Szerokość jednego segmentu fali w pikselach
    
    for (int x = left; x < right; x += step) {
        int next_x = std::min(x + step, right);

        // Obliczamy Y dla lewej i prawej krawędzi paska
        float y_left = (float)calculate_wave_y(x, surface_y_base);
        float y_right = (float)calculate_wave_y(next_x, surface_y_base);

        // Kolory (możesz tu zrobić gradient: jasny u góry, ciemny na dole)
        SDL_Color top_color = { 0, 150, 255, 180 }; // Jasny błękit
        SDL_Color bot_color = { 0, 50, 150, 220 };  // Ciemny błękit

        // Tworzymy 4 wierzchołki dla danego segmentu (trapezu)
        SDL_Vertex tl = { {(float)x - origin.x, y_left - origin.y}, top_color, {0, 0} };
        SDL_Vertex tr = { {(float)next_x - origin.x, y_right - origin.y}, top_color, {0, 0} };
        SDL_Vertex bl = { {(float)x - origin.x, (float)bottom - origin.y}, bot_color, {0, 0} };
        SDL_Vertex br = { {(float)next_x - origin.x, (float)bottom - origin.y}, bot_color, {0, 0} };

        // Dodajemy dwa trójkąty (tl-tr-bl oraz bl-tr-br)
        vertices.push_back(tl); vertices.push_back(tr); vertices.push_back(bl);
        vertices.push_back(bl); vertices.push_back(tr); vertices.push_back(br);
    }

    // Rysujemy całą wodę jednym wywołaniem GPU
    if (!vertices.empty()) {
        SDL_RenderGeometry(renderer, nullptr, vertices.data(), vertices.size(), nullptr, 0);
    }
}
